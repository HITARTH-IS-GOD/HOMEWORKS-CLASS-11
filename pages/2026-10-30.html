const form = document.getElementById('homework-form'); const dateInput = document.getElementById('homework-date'); const titleInput = document.getElementById('homework-title'); const codeInput = document.getElementById('homework-code'); const outputPanel = document.getElementById('output-panel'); const generatedOutput = document.getElementById('generated-output'); const clearFormBtn = document.getElementById('clear-form'); function escapeHtml(value) { return value .replace(/&/g, '&') .replace(//g, '>') .replace(/"/g, '"') .replace(/'/g, '''); } function generateHtml(date, title, code) { return ` ${escapeHtml(title || 'Homework')} - ${escapeHtml(date)}
Class 11 CS Homework

${escapeHtml(title || 'Homework')} - ${escapeHtml(date)}

${escapeHtml(code)}
`; } form.addEventListener('submit', (event) => { event.preventDefault(); const date = dateInput.value; const title = titleInput.value.trim() || 'Homework'; const code = codeInput.value.trim(); if (!date || !code) { alert('Please enter both the date and code.'); return; } const html = generateHtml(date, title, code); generatedOutput.value = html; outputPanel.classList.remove('hidden'); }); clearFormBtn.addEventListener('click', () => { form.reset(); generatedOutput.value = ''; outputPanel.classList.add('hidden'); });
