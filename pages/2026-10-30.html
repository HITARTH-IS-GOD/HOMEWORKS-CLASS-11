def string_1():
    s1 = input("enter string 1:")

    words1 = 0
    characters1 = 0
    alphanumchar1 = 0
    ovr_total1 = 0

    for i in s1:
        if i == " ":
            words1 += 1

    for i in s1:
        if i.isdigit() == True or i == " " or i == "," or i == ";" or i == ":" or i == "!" or i == "?":
            characters1 += 1

    for i in s1:
        if i.isdigit() == True or i.isalpha() == True:
            alphanumchar1 += 1

    for i in s1:
        ovr_total1 += 1

    if ovr_total1 > 0:
        per_char_alphanum1 = (alphanumchar1 / ovr_total1) * 100
    else:
        per_char_alphanum1 = 0

    print(words1)
    print(characters1)
    print(per_char_alphanum1)
